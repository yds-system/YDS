# 디자인 토큰

디자인 토큰은 색상·타이포·간격·모서리 같은 시각적 결정을 코드로 표현한 이름 붙은 변수입니다.
컴포넌트는 값을 직접 쓰지 않고 토큰 이름만 참조합니다.

---

## 네이밍 규칙

```
--{분류}-{대상}-{역할}-{상태}
```

| 계층 | 설명 | 예시 |
|------|------|------|
| **프리미티브** | 원재료. 값 그 자체. 컴포넌트에서 직접 쓰지 않는다. | `--blue-50`, `--gray-90`, `--spacing-16` |
| **시멘틱** | 용도 기반. 반드시 프리미티브를 `var()`로 참조한다. | `--bg-brand`, `--txt-primary`, `--gap-m` |
| **테마 오버라이드** | `[data-theme]`에서 시멘틱 토큰 일부를 교체한다. | `--bg-brand: var(--green-50)` |

**프리미티브는 물감, 시멘틱은 용도가 적힌 물감통.**
색칠할 때는 물감통(시멘틱)을 집어든다. 내부에 무슨 물감(프리미티브)이 들어 있는지는 몰라도 된다.
테마를 바꿀 때는 물감통에 다른 물감을 채워 넣으면(오버라이드) 모든 컴포넌트가 자동으로 바뀐다.

**장식 vs UI 사용 구분**

- **아이콘·일러스트·차트 등 장식 요소** → 프리미티브 직접 참조 허용 (`--blue-50`, `--green-30` 등)
- **버튼·텍스트·배경·테두리 등 UI 요소** → 반드시 시멘틱 참조 (`--bg-brand`, `--txt-primary` 등)

export const TypoStyleTable = ({ tokens }) => {
  const chipStyle = { background: '#f1f3f5', padding: '1px 5px', borderRadius: '3px', fontSize: '12px', fontFamily: 'monospace', whiteSpace: 'nowrap' };
  const chip = (txt) => h('code', { style: chipStyle }, txt);
  const th = (txt) => h('th', { style: { padding: '5px 8px', background: '#f8f8f8', textAlign: 'left', borderBottom: '2px solid #e0e0e0', whiteSpace: 'nowrap', fontSize: '12px' } }, txt);
  const td = (content, extra = {}) => h('td', { style: { padding: '4px 8px', borderBottom: '1px solid #eee', verticalAlign: 'middle', fontSize: '12px', ...extra } }, content);

  const sample = (t) => {
    const [szStr, wStr, lhStr] = (t.valuePC || '').split(' / ');
    const sz = parseFloat(szStr) * 10 + 'px';
    const fw = wStr === 'bold' ? '700' : wStr === '500' ? '500' : '400';
    return h('span', { style: { fontSize: sz, fontWeight: fw, lineHeight: '1.5', color: '#1e2124', letterSpacing: 0 } }, '가나다 Aa');
  };

  return h('table', { style: { width: '100%', borderCollapse: 'collapse' } },
    h('thead', null,
      h('tr', null, th('견본'), th('CSS 변수 (font-size)'), th('PC 값 (크기/굵기/행간)'), th('모바일'), th('용도'))
    ),
    h('tbody', null,
      ...tokens.map(t =>
        h('tr', { key: t.css },
          td(sample(t)),
          td(chip(t.css)),
          td(h('code', { style: { ...chipStyle, background: 'transparent' } }, t.valuePC)),
          td(t.valuePC === t.valueMobile
            ? h('span', { style: { color: '#aaa', fontSize: '11px' } }, '동일')
            : h('code', { style: { ...chipStyle, background: '#fff8e0' } }, t.valueMobile)
          ),
          td(t.usage || '—')
        )
      )
    )
  );

    return h('div', { style: { width: '40px', height: '40px', background: '#ecf2fe', border: '2px solid #256ef4', borderRadius: r, flexShrink: 0 } });
  };

  return h('table', { style: { width: '100%', borderCollapse: 'collapse' } },
    h('thead', null,
      h('tr', null, th('견본'), th('CSS 변수'), th('참조 프리미티브'), th('값'), th('용도'), th('KRDS 원본'))
    ),
    h('tbody', null,
      ...tokens.map(t =>
        h('tr', { key: t.css },
          td(box(t.value)),
          td(chip(t.css)),
          td(chip(t.ref)),
          td(h('code', { style: { ...chipStyle, background: 'transparent' } }, t.value)),
          td(t.usage || '—'),
          td(h('code', { style: { ...chipStyle, background: 'transparent', color: '#888' } }, t.krds))
        )
      )
    )
  );
};
